# javpipe 定时任务示例：crontab -e 贴进去，改掉路径
# 建议 scrape 与 rescue 错开几分钟，避免同一分钟抢同一目录

# 每 15 分钟：扫下载目录 → 刮削 → 入库
*/15 * * * * cd /path/to/javpipe && .venv/bin/javpipe scrape >> /var/log/javpipe.log 2>&1

# 每小时第 5 分钟：救失败区（刮削没过但片子还在的）
5 * * * * cd /path/to/javpipe && .venv/bin/javpipe rescue >> /var/log/javpipe.log 2>&1

# 每天 04:10：体检，异常时退出码非 0（可接监控）
10 4 * * * cd /path/to/javpipe && .venv/bin/javpipe doctor >> /var/log/javpipe.log 2>&1
